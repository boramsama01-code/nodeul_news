SECTION 2 — RELEVANCE FILTER (implement exactly as written)
typescriptfunction isRelevant(title: string, content: string): boolean {
  const text = (title + " " + content);

  // STEP 1: Hard include — if any of these exist, always return true immediately
  if (
    text.includes("노들섬") ||
    text.includes("노들 예술섬") ||
    text.includes("예술섬 노들")
  ) {
    return true;
  }

  // STEP 2: Conditional include — "노들" present but no exclusion patterns
  if (text.includes("노들")) {
    const exclusionPatterns = [
      "노들역", "노들로1", "노들로2", "노들로3", "노들로4", "노들로5",
      "노들로6", "노들로7", "노들로8", "노들로9", "노들로0",
      "노들나루공원", "노들나루역"
    ];
    const hasExclusion = exclusionPatterns.some(e => text.includes(e));
    return !hasExclusion;
  }

  return false;
}
CRITICAL: Step 1 runs first and always wins. An article containing "노들섬" is ALWAYS included even if it also contains "노들역", "노들나루역", or any other exclusion pattern. Exclusion patterns only apply to articles that did NOT pass Step 1.

SECTION 4 — DEDUPLICATION
typescriptfunction isDuplicate(url: string, title: string, publishedAt: Date, existingArticles: Set<string>): boolean {
  // Primary check: exact URL match
  const normalizedUrl = url.toLowerCase().trim().replace(/\/$/, "");
  if (existingArticles.has(normalizedUrl)) return true;

  // Secondary check: title + date fingerprint
  const dateStr = publishedAt.toISOString().split("T")[0];
  const fingerprint = `${title.trim().replace(/\s+/g, "")}__${dateStr}`;
  if (existingArticles.has(fingerprint)) return true;

  return false;
}
Load all existing URLs and fingerprints from DB into memory at the start of each crawl job. Update the in-memory set as new articles are saved so duplicates within the same crawl batch are also caught.

SECTION 6 — CRAWL JOB ORCHESTRATION
typescriptasync function runCrawlJob(jobId: string, startDate: string, endDate: string) {
  // 1. Load existing URLs and fingerprints from DB into memory
  const existingSet = new Set<string>();
  const existingArticles = await prisma.article.findMany({ select: { url: true, title: true, publishedAt: true } });
  for (const a of existingArticles) {
    existingSet.add(a.url.toLowerCase().trim().replace(/\/$/, ""));
    const fp = `${a.title.trim().replace(/\s+/g, "")}__${a.publishedAt.toISOString().split("T")[0]}`;
    existingSet.add(fp);
  }

  let totalCollected = 0;
  let totalDuplicates = 0;
  const sources = ["네이버API", "구글RSS", "다음뉴스", "BigKinds", "RSS피드들", "언론사검색", "네이버검색"];
  
  for (let i = 0; i < sources.length; i++) {
    await prisma.crawlJob.update({ where: { id: jobId }, data: { currentSource: sources[i], progress: Math.round((i / sources.length) * 90) } });
    
    // run each source, catch errors per source
    try {
      const results = await runSource(sources[i], startDate, endDate);
      for (const article of results) {
        if (!isRelevant(article.title, article.content)) continue;
        if (isDuplicate(article.url, article.title, article.publishedAt, existingSet)) {
          totalDuplicates++;
          continue;
        }
        // Add to set before saving to catch within-batch duplicates
        existingSet.add(article.url.toLowerCase().trim().replace(/\/$/, ""));
        const fp = `${article.title.trim().replace(/\s+/g, "")}__${article.publishedAt.toISOString().split("T")[0]}`;
        existingSet.add(fp);
        
        await prisma.article.create({ data: article });
        totalCollected++;
        await prisma.crawlJob.update({ where: { id: jobId }, data: { collected: totalCollected, duplicates: totalDuplicates } });
      }
    } catch (err) {
      console.error(`Source ${sources[i]} failed:`, err);
      // Continue to next source
    }
  }
  
  await prisma.crawlJob.update({ where: { id: jobId }, data: { status: "done", progress: 100, total: totalCollected + totalDuplicates } });
}

typescript// Backend: POST /api/articles/:id/analyze
async function analyzeArticle(id: number) {
  const article = await prisma.article.findUnique({ where: { id } });
  if (!article) throw new Error("Article not found");

  const response = await anthropic.messages.create({
    model: "claude-sonnet-4-20250514",
    max_tokens: 100,
    messages: [{
      role: "user",
      content: `다음 기사를 분석하여 노들섬에 대해 부정적인 내용인지 판단하세요.
부정적인 기사란: 노들섬 개발/운영/정책에 대한 비판, 문제점 지적, 반대 의견, 부정적 사건 보도 등을 포함합니다.
반드시 JSON만 반환하세요. 다른 텍스트 없이: {"isNegative": true} 또는 {"isNegative": false}

제목: ${article.title}
내용: ${article.content.slice(0, 500)}`
    }]
  });

  const text = response.content[0].type === "text" ? response.content[0].text : "";
  const cleaned = text.replace(/```json|```/g, "").trim();
  const parsed = JSON.parse(cleaned);

  return prisma.article.update({
    where: { id },
    data: { isNegative: parsed.isNegative }
  });
}

SECTION 10 — ERROR HANDLING & RESILIENCE
Implement these safeguards throughout:

Every external HTTP request: wrap in try/catch, set timeout (axios: timeout: 10000), retry up to 2 times on network error with 1s delay between retries
RSS feed failures: if a feed returns non-XML or 404 or times out, log [RSS SKIP] feedName: reason and continue
Date parsing failures: if publishedAt cannot be parsed, skip the article (do not save with wrong date)
URL validation: before saving, check that url starts with http and is a valid URL. Discard invalid URLs.
Prisma unique constraint on url: wrap article creation in try/catch for P2002 (unique violation) — treat as duplicate and increment duplicate counter
BigKinds auth failure: if response status 401/403 or error contains "login", skip gracefully
Google News redirect failures: if redirect chain exceeds 10 hops or times out, skip that article
Naver API rate limit: if response status 429, wait 2 seconds and retry once
Job crash recovery: if the Node process crashes during a crawl, the CrawlJob record stays as "running" — add a startup check that sets any "running" jobs older than 10 minutes to "error" status on server start
Content encoding: use axios.get(url, { responseType: "arraybuffer" }) and detect encoding from Content-Type header, convert with iconv-lite if needed (many Korean news sites use EUC-KR)
Cheerio HTML parsing: always check if selector returned any elements before accessing — never assume selectors will work
